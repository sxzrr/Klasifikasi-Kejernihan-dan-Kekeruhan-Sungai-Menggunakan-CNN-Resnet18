# Improvements:

# Unfreeze backbone untuk fine-tuning
for param in model.parameters():
    param.requires_grad = True  # Enable gradients untuk semua parameters

# Optimizer dengan weight decay (L2 regularization)
# Weight decay penalizes large weights → prevent memorization
optimizer = optim.Adam(model.parameters(), lr=0.00005, weight_decay=1e-4)

# Learning rate scheduler
scheduler = optim.lr_scheduler.ReduceLROnPlateau(
    optimizer, mode='max', factor=0.5, patience=3
)

# Variables untuk Early Stopping
history_stage2 = {'train_loss': [], 'train_acc': [], 'val_loss': [], 'val_acc': []}
best_val_acc_stage2 = 0
patience = 5  # Stop jika val_acc tidak improve selama 5 epochs
patience_counter = 0
best_model_state = None

print("\n" + "="*60)
print("Stage 2: Fine-tuning Backbone with Early Stopping (10 epochs)")
print("="*60)
print("Early Stopping: patience=5 epochs")
print("Weight Decay: L2 regularization = 1e-4")
print("Learning Rate: 0.00005 (reduced from 0.0001)")
print("="*60 + "\n")

# Training loop (max 10 epochs, tapi akan stop lebih awal jika tidak improve)
for epoch in range(10):  # Reduced from 20 to 10 epochs
    train_loss, train_acc = train_epoch(model, train_loader, criterion, optimizer, device)
    val_loss, val_acc = validate(model, val_loader, criterion, device)
    
    # Store history
    history_stage2['train_loss'].append(train_loss)
    history_stage2['train_acc'].append(train_acc)
    history_stage2['val_loss'].append(val_loss)
    history_stage2['val_acc'].append(val_acc)
    
    # Learning rate scheduler step
    scheduler.step(val_acc)
    
    # Early Stopping Logic
    if val_acc > best_val_acc_stage2:
        # Improvement detected - reset patience counter
        best_val_acc_stage2 = val_acc
        patience_counter = 0
        
        # Save best model state
        best_model_state = model.state_dict().copy()
        
        print(f"Epoch {epoch+1:2d}/10: "
              f"Train Loss={train_loss:.4f} | Train Acc={train_acc:.4f} | "
              f"Val Loss={val_loss:.4f} | Val Acc={val_acc:.4f} | "
              f"⭐ BEST")
    else:
        # No improvement - increment patience counter
        patience_counter += 1
        
        print(f"Epoch {epoch+1:2d}/10: "
              f"Train Loss={train_loss:.4f} | Train Acc={train_acc:.4f} | "
              f"Val Loss={val_loss:.4f} | Val Acc={val_acc:.4f} | "
              f"⏱️  Patience {patience_counter}/{patience}")
        
        # Early stopping: stop jika patience exceeded
        if patience_counter >= patience:
            print(f"\n✅ Early stopping at epoch {epoch+1}")
            print(f"Best validation accuracy: {best_val_acc_stage2:.4f}")
            break

# Restore best model state before stopping
if best_model_state is not None:
    model.load_state_dict(best_model_state)
    print(f"✅ Restored best model (val_acc={best_val_acc_stage2:.4f})")

# Save final model
torch.save(model.state_dict(), model_save_path)
print(f"✅ Model saved to {model_save_path}")
